    <script>
        // DOM Elements
        const generateNewsButton = document.getElementById('generateNewsButton');

        const copyTextButton = document.getElementById('copyTextButton');

        // Message Box Elements
        const messageBox = document.getElementById('messageBox');
        const messageText = document.getElementById('messageText');
        const closeMessageButton = document.getElementById('closeMessageButton');

        // Webhook URLs
        const newsWebhookUrl = 'https://agentify360.app.n8n.cloud/webhook/1ad022c2-e67d-4027-a66f-d06d637dc50c/chat'; // UPDATED URL
        const imageWebhookUrl = 'https://agentify360.app.n8n.cloud/webhook/76840a22-558d-4fae-9f51-aadcd7c3fb7f';

        let currentObjectUrl = null; 

        // Function to generate a simple unique Chat ID
        function generateChatID() {
            return 'chat_' + Date.now() + '_' + Math.random().toString(36).substring(2, 9);
        }

        // Function to show a message in the custom message box
        function showMessage(message, isError = false) {
            messageText.textContent = message;
            messageBox.className = `fixed top-5 right-5 p-4 rounded-lg shadow-lg text-sm ${isError ? 'bg-red-600 text-white' : 'bg-blue-600 text-white'}`;
            messageBox.style.display = 'block';
            setTimeout(() => {
                messageBox.style.display = 'none';
            }, 7000);
        }

        closeMessageButton.addEventListener('click', () => {
            messageBox.style.display = 'none';
        });

        /**
         * Extracts news text from a potentially JSON-formatted string,
         * cleans some Markdown/HTML, and converts basic Markdown to HTML for display.
         * @param {string} outputString The string received from the webhook (data.output).
         * @returns {string} The HTML formatted string for display.
         */
        function extractAndFormatNewsTextToHTML(outputString) {
            if (typeof outputString !== 'string') {
                console.warn("extractAndFormatNewsTextToHTML: input is not a string", outputString);
                return '<p class="text-red-500">[Invalid input to text formatter]</p>';
            }

            let newsContent = outputString; 
            try {
                const firstParse = JSON.parse(outputString);
                if (firstParse && typeof firstParse["AI News piece"] === 'string') {
                    newsContent = firstParse["AI News piece"];
                } else {
                    const commonKeys = ["article_text", "content", "text", "summary", "message", "news_content", "article"];
                    let found = false;
                    for (const key of commonKeys) {
                        if (typeof firstParse[key] === 'string') {
                            newsContent = firstParse[key];
                            found = true;
                            break;
                        }
                    }
                    if (!found) {
                        console.warn("Parsed outputString as JSON, but no recognized news text field found. Raw JSON:", firstParse);
                         if (firstParse.urls && Array.isArray(firstParse.urls)) { // Example: if JSON contains a list of URLs
                             return `<p class="text-red-500">[Received structured data with ${firstParse.urls.length} URLs. Main news text not found.]</p><p>${firstParse.urls.map(url => `<a href="${url}" target="_blank" rel="noopener noreferrer">${url}</a>`).join("<br>")}</p>`;
                        }
                        return '<p class="text-red-500">[Structured data received, but main news text could not be extracted. Check webhook output (data.output).]</p>';
                    }
                }
            } catch (e) {
                console.log("outputString was not a stringified JSON. Treating as direct text for HTML formatting.", e.message);
            }
            
            let htmlText = String(newsContent);

            // 1. Handle escaped newlines first
            htmlText = htmlText.replace(/\\n/g, '\n');

            // 2. Basic HTML escaping for security (important before injecting as HTML)
            const escapeDiv = document.createElement('div');
            escapeDiv.textContent = htmlText;
            htmlText = escapeDiv.innerHTML; // Now htmlText is HTML-escaped plain text

            // 3. Convert Markdown bold and italics to HTML
            htmlText = htmlText.replace(/\*\*(.*?)\*\*/g, '<b>$1</b>'); 
            htmlText = htmlText.replace(/__(.*?)__/g, '<b>$1</b>');   
            htmlText = htmlText.replace(/\*(.*?)\*/g, '<i>$1</i>');     
            htmlText = htmlText.replace(/_(.*?)_/g, '<i>$1</i>');       
            
            // 4. Convert specific phrases like "Top article:" and "Next 7 articles:" to bolded paragraphs
            htmlText = htmlText.replace(/^Top article:/gim, '<p class="mt-2 mb-1"><b>Top article:</b></p>');
            htmlText = htmlText.replace(/^Next 7 articles:/gim, '<p class="mt-2 mb-1"><b>Next 7 articles:</b></p>');
            // Convert general Markdown headings
            htmlText = htmlText.replace(/^#+\s+(.*)/gm, '<h3>$1</h3>'); 

            // 5. Convert Markdown links [text](url) to HTML links
            htmlText = htmlText.replace(/\[([^\]]+)\]\(([^)]+)\)/g, '<a href="$2" target="_blank" rel="noopener noreferrer" class="text-blue-600 hover:text-blue-800 underline">$1</a>');

            // 6. Convert standalone URLs to HTML links
            const urlRegex = /(\b(https?|ftp|file):\/\/[-A-Z0-9+&@#\/%?=~_|!:,.;]*[-A-Z0-9+&@#\/%=~_|])(?![^<]*<\/a>|[^<>]*>)/ig;
            htmlText = htmlText.replace(urlRegex, '<a href="$1" target="_blank" rel="noopener noreferrer" class="text-blue-600 hover:text-blue-800 underline">$1</a>');
            
            // 7. Convert actual newlines to <br> for HTML display, and handle paragraph-like spacing
            // Split by double newlines first to create paragraphs, then single newlines to <br>
            htmlText = htmlText.split(/\n\s*\n/).map(paragraph => {
                if (paragraph.trim() === '') return '';
                // Don't wrap lines that are already h3 or start with our bolded phrases in <p>
                if (paragraph.trim().startsWith('<h3') || paragraph.trim().startsWith('<p class="mt-2 mb-1"><b>')) {
                    return paragraph.replace(/\n/g, '<br>\n');
                }
                return '<p>' + paragraph.replace(/\n/g, '<br>\n') + '</p>';
            }).join('');
            
            // Remove empty paragraphs that might have been created
            htmlText = htmlText.replace(/<p><\/p>/g, '');
            htmlText = htmlText.replace(/<p><br\s*\/?>\s*<\/p>/gi, ''); 

            return htmlText;
        }
        
        // Event Listener for "Generate Daily News" button
        generateNewsButton.addEventListener('click', async () => {
            const chatID = generateChatID();
            generateNewsButton.disabled = true;
            generateNewsButton.classList.add('opacity-50', 'cursor-not-allowed');
            newsLoader.style.display = 'inline-block';
            newsGenerationStatus.textContent = 'Generating news content...';
            newsDisplayArea.innerHTML = ''; 

            try {
                const response = await fetch(newsWebhookUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json', },
                    body: JSON.stringify({ message: 'generate daily news', chatID: chatID }),
                });

                if (!response.ok) {
                    let errorDetails = `Server responded with status ${response.status} ${response.statusText}.`;
                    try { 
                        const errorData = await response.json(); 
                        errorDetails += ` Response: ${JSON.stringify(errorData)}`; 
                    } catch (e) { /* Failed to parse error response body */ }
                    throw new Error(errorDetails);
                }
                
                const data = await response.json(); 

                if (data && typeof data.output === 'string') {
                    const formattedHTML = extractAndFormatNewsTextToHTML(data.output);
                    newsDisplayArea.innerHTML = formattedHTML; 
                    
                    if (formattedHTML.includes("[Structured data received") || formattedHTML.includes("[Invalid input")) {
                        newsGenerationStatus.textContent = 'News data received, but main text not found or invalid.';
                        showMessage('Received structured data or invalid input. Main news text might be missing or in an unexpected format.', true);
                        generateImageButton.style.display = 'none';
                        copyTextButton.style.display = 'none';
                    } else if (newsDisplayArea.textContent.trim() === "") { 
                         newsGenerationStatus.textContent = 'News content received, but it appears to be empty after formatting.';
                         showMessage('News content is empty after formatting. Check webhook output.', true);
                         generateImageButton.style.display = 'none';
                         copyTextButton.style.display = 'none';
                    } else {
                        newsGenerationStatus.textContent = 'News content generated and formatted.';
                        showMessage('News content loaded and formatted!', false);
                        generateImageButton.style.display = 'flex'; 
                        copyTextButton.style.display = 'inline-flex'; 
                    }
                } else if (data && data.output && typeof data.output !== 'string') {
                    console.warn('News webhook data.output was not a string but an object. Attempting to find "AI News piece" directly or stringify and process.', data.output);
                    let newsTextToFormat = "";
                    if (data.output["AI News piece"] && typeof data.output["AI News piece"] === 'string') {
                        newsTextToFormat = data.output["AI News piece"];
                    } else {
                        newsTextToFormat = JSON.stringify(data.output); 
                    }
                    const formattedHTML = extractAndFormatNewsTextToHTML(newsTextToFormat);
                    newsDisplayArea.innerHTML = formattedHTML;
                    newsGenerationStatus.textContent = 'News content (complex type) processed.';
                    generateImageButton.style.display = 'flex'; 
                    copyTextButton.style.display = 'inline-flex';
                } else {
                    console.warn('News webhook response did not contain a "data.output" string or it was empty.', data);
                    newsDisplayArea.innerHTML = '<p class="text-red-500">[News content not found in webhook response (data.output missing, not a string, or empty). Check console.]</p>';
                    newsGenerationStatus.textContent = 'News content missing in response.';
                    showMessage('News content not found in the expected part of the webhook response.', true);
                    generateImageButton.style.display = 'none';
                    copyTextButton.style.display = 'none';
                }
                
                newsDisplayArea.focus(); 
                const hasValidText = newsDisplayArea.textContent.trim() !== "" && !newsDisplayArea.textContent.trim().startsWith("[");
                generateImageButton.disabled = !hasValidText;
                copyTextButton.disabled = !hasValidText;
                generateImageButton.classList.toggle('opacity-50', !hasValidText);
                generateImageButton.classList.toggle('cursor-not-allowed', !hasValidText);
                copyTextButton.classList.toggle('opacity-50', !hasValidText);
                copyTextButton.classList.toggle('cursor-not-allowed', !hasValidText);


            } catch (error) {
                console.error('Full error object from news webhook:', error); 
                let errorMessage = (error instanceof Error) ? error.message : String(error);
                newsGenerationStatus.textContent = `Error: ${errorMessage}. Check console.`;
                showMessage(`Failed to fetch news: ${errorMessage}. Please try again.`, true);
                newsDisplayArea.innerHTML = '<p class="text-red-500">[Failed to load news content due to an error.]</p>'; 
                generateImageButton.style.display = 'none';
                copyTextButton.style.display = 'none';
            } finally {
                generateNewsButton.disabled = false; 
                generateNewsButton.classList.remove('opacity-50', 'cursor-not-allowed');
                newsLoader.style.display = 'none';
            }
        });

        // Event listener for "Copy Text" button
        copyTextButton.addEventListener('click', () => {
            const textToCopy = newsDisplayArea.textContent; 
            if (!textToCopy) {
                showMessage('Nothing to copy.', true);
                return;
            }
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(textToCopy)
                    .then(() => {
                        showMessage('Text copied to clipboard!', false);
                    })
                    .catch(err => {
                        console.error('Failed to copy text: ', err);
                        showMessage('Failed to copy text. See console for details.', true);
                    });
            } else {
                try {
                    const textArea = document.createElement("textarea");
                    textArea.value = textToCopy;
                    textArea.style.position = "fixed"; 
                    document.body.appendChild(textArea);
                    textArea.focus();
                    textArea.select();
                    document.execCommand('copy');
                    document.body.removeChild(textArea);
                    showMessage('Text copied to clipboard (fallback)!', false);
                } catch (err) {
                    console.error('Fallback copy failed: ', err);
                    showMessage('Failed to copy text using fallback. Your browser might not support this feature.', true);
                }
            }
        });


        // Event Listener for "Generate Image" button (Shows Google Drive Link)
        generateImageButton.addEventListener('click', async () => {
            const newsText = newsDisplayArea.textContent.trim(); 

            if (!newsText || newsText.startsWith("[")) {
                showMessage('Valid news text is required to generate an image link.', true);
                newsDisplayArea.focus();
                return;
            }

            generateImageButton.disabled = true;
            generateImageButton.classList.add('opacity-50', 'cursor-not-allowed');
            downloadImageLink.style.display = 'none'; 
            imageLoader.style.display = 'inline-block';
            imageGenerationStatus.textContent = 'Generating image link...';
            imageLinkDisplay.innerHTML = `<span class="text-gray-500">Generating link...</span>`; 
            if (imageDisplaySectionDiv) imageDisplaySectionDiv.style.display = 'none'; 

            let response; 
            try {
                response = await fetch(imageWebhookUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json', }, 
                    body: JSON.stringify({ news_text: newsText }),
                });

                if (!response.ok) {
                    let errorDetails = `Image link generation failed: ${response.status} ${response.statusText}.`;
                    try {
                        const serverErrorText = await response.text(); 
                        console.error("Server error response text:", serverErrorText); 
                        errorDetails += ` Server says: ${serverErrorText.substring(0, 200)}`; 
                    } catch (textError) {
                        console.error("Could not read server error response body:", textError);
                        errorDetails += " Could not read error response body.";
                    }
                    throw new Error(errorDetails);
                }
                
                const contentType = response.headers.get("content-type");
                console.log("Image Webhook Content-Type:", contentType); 

                if (contentType && contentType.includes("application/json")) {
                    const jsonData = await response.json();
                    console.log("Image Webhook JSON response:", JSON.stringify(jsonData, null, 2)); 

                    if (jsonData.webViewLink && typeof jsonData.webViewLink === 'string') {
                        imageLinkDisplay.innerHTML = `Image Link: <a href="${jsonData.webViewLink}" target="_blank" rel="noopener noreferrer" class="break-all">${jsonData.webViewLink}</a>`;
                        imageGenerationStatus.textContent = 'Image link generated successfully!';
                        showMessage('Image link generated!', false);
                    } else {
                        console.warn("JSON response received, but 'webViewLink' key was missing or not a string.", jsonData);
                        imageLinkDisplay.innerHTML = `<span class="text-red-500">Error: 'webViewLink' not found in response. Check console.</span>`;
                        throw new Error("'webViewLink' not found in JSON response. Check console for full response.");
                    }
                } else {
                     const responseText = await (response.bodyUsed ? response.clone().text() : response.text());
                     console.warn("Image webhook response was not JSON. Content-Type:", contentType, "Body:", responseText.substring(0,500));
                     imageLinkDisplay.innerHTML = `<span class="text-red-500">Error: Expected JSON response with 'webViewLink', but received different format. Check console.</span>`;
                     throw new Error(`Expected JSON response, but received Content-Type: ${contentType}.`);
                }

            } catch (error) {
                console.error('Full error object from image webhook:', error); 
                
                let errorMessage = 'An unknown error occurred during image link generation.';
                if (error instanceof Error) { 
                    errorMessage = error.message;
                } else if (typeof error === 'object' && error !== null) { 
                    try { errorMessage = JSON.stringify(error); } catch (e) { /* ignore stringify error */ }
                } else if (error) { 
                    errorMessage = String(error);
                }

                if (error instanceof TypeError && error.message.toLowerCase().includes("failed to fetch")) {
                     errorMessage = 'Network error: Could not connect to the image service. Check CORS or server status.';
                     console.error("A 'TypeError: Failed to fetch' occurred during image link generation. Check network connection, CORS policy, and server availability.");
                }
                
                imageGenerationStatus.textContent = `Error: ${errorMessage}.`; 
                imageLinkDisplay.innerHTML = `<span class="text-red-500">Error generating link: ${errorMessage}</span>`;
                showMessage(`Image link generation failed: ${errorMessage}`, true);
            } finally {
                generateImageButton.disabled = false;
                generateImageButton.classList.remove('opacity-50', 'cursor-not-allowed');
                imageLoader.style.display = 'none';
            }
        });

        // Initial state setup
        generateImageButton.style.display = 'none';
        copyTextButton.style.display = 'none';

    </script>
